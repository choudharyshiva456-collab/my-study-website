/* =========================

let timeLeft=300;
let timerInterval=null;

function updateTimer(){

let m=Math.floor(timeLeft/60);
let s=timeLeft%60;

document.getElementById("timerDisplay").innerText=
String(m).padStart(2,"0")+":"+
String(s).padStart(2,"0");

}

function startTimer(){

if(timerInterval!==null)return;

document.getElementById("timerMessage").innerText="";

timerInterval=setInterval(()=>{

timeLeft--;

updateTimer();

if(timeLeft<=0){

clearInterval(timerInterval);
timerInterval=null;

document.getElementById("timerMessage").innerText=
"🎉 Study session completed!";

let sessions=Number(localStorage.getItem("sessions")||0)+1;
let minutes=Number(localStorage.getItem("minutes")||0)+5;

localStorage.setItem("sessions",sessions);
localStorage.setItem("minutes",minutes);

updateDashboard();

}

},1000);

}

function pauseTimer(){

clearInterval(timerInterval);
timerInterval=null;

document.getElementById("timerMessage").innerText=
"⏸️ Timer paused.";

}

function resetTimer(){

clearInterval(timerInterval);
timerInterval=null;

timeLeft=300;

updateTimer();

document.getElementById("timerMessage").innerText="";

}

function addNote(){

let subject=document.getElementById("noteSubject").value;
let title=document.getElementById("noteTitle").value.trim();
let text=document.getElementById("noteText").value.trim();

if(!title||!text){

alert("कृपया title और note दोनों लिखें।");
return;

}

let notes=JSON.parse(localStorage.getItem("notes")||"[]");

notes.push({
subject:subject,
title:title,
text:text,
date:new Date().toLocaleDateString()
});

localStorage.setItem("notes",JSON.stringify(notes));

document.getElementById("noteTitle").value="";
document.getElementById("noteText").value="";

displayNotes();
updateDashboard();

}

function displayNotes(){

let notes=JSON.parse(localStorage.getItem("notes")||"[]");

let list=document.getElementById("notesList");

list.innerHTML="";

notes.slice().reverse().forEach((note,index)=>{

let card=document.createElement("div");

card.className="note-card";

card.innerHTML=`
<span class="badge">${note.subject}</span>
<h3>${escapeHTML(note.title)}</h3>
<p>${escapeHTML(note.text)}</p>
<small>📅 ${note.date}</small>
<br>
<button class="danger"
onclick="deleteNote(${notes.length-1-index})">
🗑️ Delete
</button>
`;

list.appendChild(card);

});

}

function deleteNote(index){

let notes=JSON.parse(localStorage.getItem("notes")||"[]");

notes.splice(index,1);

localStorage.setItem("notes",JSON.stringify(notes));

displayNotes();
updateDashboard();

}

function escapeHTML(text){

let div=document.createElement("div");
div.textContent=text;
return div.innerHTML;

}


/* =========================
   MCQ
========================= */

function loadSubjectMCQ(){

currentSubject=document.getElementById("mcqSubject").value;

currentQuestions=
[...questionBank[currentSubject]];

currentIndex=0;
testSubmitted=false;

document.getElementById("result").style.display="none";

renderMCQ();

}

function renderMCQ(){

let container=document.getElementById("mcqContainer");

container.innerHTML="";

currentQuestions.forEach((item,index)=>{

let div=document.createElement("div");

div.className="mcq";

if(index===currentIndex){
div.classList.add("active");
}

let options="";

item.o.forEach((option,i)=>{

options+=`
<label class="option">
<input type="radio"
name="question${index}"
value="${i}">
${String.fromCharCode(65+i)}. ${option}
</label>
`;

});

div.innerHTML=`
<h3>Q${index+1}. ${item.q}</h3>
${options}
`;

container.appendChild(div);

});

updateCounter();

}

function showQuestion(){

document.querySelectorAll(".mcq").forEach((q,i)=>{

q.classList.toggle(
"active",
i===currentIndex
);

});

updateCounter();

}

function updateCounter(){

document.getElementById("counter").innerText=
"Q "+(currentIndex+1)+" / "+currentQuestions.length;

}

function nextQuestion(){

if(currentIndex<currentQuestions.length-1){

currentIndex++;
showQuestion();

}

}

   TIMED TEST

function startTimedTest(){

randomTest();

timedSeconds=60;

document.getElementById("testTimer").innerText=
"⏳ Time Left: 60 seconds";

clearInterval(timedInterval);

timedInterval=setInterval(()=>{

timedSeconds--;

document.getElementById("testTimer").innerText=
"⏰ Time Up!";

submitTest();

}

},1000);

}


/* =========================
   SUBMIT
========================= */

function submitTest(){

if(testSubmitted)return;

testSubmitted=true;

clearInterval(timedInterval);

let score=0;
let answered=0;

currentQuestions.forEach((item,index)=>{

let selected=document.querySelector(
`input[name="question${index}"]:checked`
);

if(selected){

answered++;

if(Number(selected.value)===item.a){
score++;
}

}

});

let total=currentQuestions.length;

let percentage=Math.round(score/total*100);

let wrong=answered-score;

let performance="";

if(percentage===100)
performance="🏆 Perfect Score!";

else if(percentage>=80)
performance="🌟 Excellent Work!";

else if(percentage>=60)
performance="👍 Good Job!";

else
performance="📚 More Revision Needed.";


document.getElementById("score").innerText=
score+" / "+total;

document.getElementById("details").innerText=
"✅ Correct: "+score+
" | ❌ Wrong: "+wrong+
" | 📝 Attempted: "+answered+
" | 📊 Percentage: "+percentage+"%";

document.getElementById("performance").innerText=
performance;

document.getElementById("result").style.display="block";


/* SAVE STATS */

let tests=Number(localStorage.getItem("tests")||0)+1;

let solved=Number(localStorage.getItem("solved")||0)+answered;

let best=Number(localStorage.getItem("best")||0);

if(percentage>best)best=percentage;

localStorage.setItem("tests",tests);
localStorage.setItem("solved",solved);
localStorage.setItem("best",best);


/* HISTORY */

let history=JSON.parse(
localStorage.getItem("history")||"[]"
);

history.push({
subject:currentSubject,
score:score,
total:total,
percentage:percentage,
date:new Date().toLocaleString()
});

localStorage.setItem(
"history",
JSON.stringify(history)
);

updateDashboard();
displayHistory();

document.getElementById("result").scrollIntoView();

}

   HISTORY

function displayHistory(){

let history=JSON.parse(
localStorage.getItem("history")||"[]"
);

let list=document.getElementById("historyList");

list.innerHTML="";

if(history.length===0){

list.innerHTML="<p>अभी कोई test history नहीं है।</p>";

}

history.slice().reverse().forEach(item=>{

let div=document.createElement("div");

div.className="history-item";

div.innerHTML=`
<span class="badge">${item.subject}</span>
<strong>${item.score}/${item.total}</strong>
— ${item.percentage}%
<br>
<small>📅 ${item.date}</small>
`;

list.appendChild(div);

});

}
