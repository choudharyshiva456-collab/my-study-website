        /* =========================
           STORAGE HELPERS
        ========================= */

        function getData(key, defaultValue) {

            let data =
                localStorage.getItem(key);

            if (data === null) {

                return defaultValue;

            }

            try {

                return JSON.parse(data);

            } catch {

                return defaultValue;

            }

        }


        function saveData(key, value) {

            localStorage.setItem(

        /* =========================
           TIMER
        ========================= */

        const TIMER_START =
            5 * 60;

        let timeLeft =
            getData(
                "timerTimeLeft",
                TIMER_START
            );

        let timerInterval = null;


        function updateTimer() {

            let minutes =
                Math.floor(
                    timeLeft / 60
                );

            let seconds =
                timeLeft % 60;


            minutes =
                String(minutes)
                    .padStart(2, "0");

            seconds =
                String(seconds)
                    .padStart(2, "0");


            document.getElementById(
                "timerDisplay"
            ).innerText =
                minutes + ":" + seconds;

        }


        function saveTimer() {

            saveData(
                "timerTimeLeft",
                timeLeft
            );

        }


        function startTimer() {

            if (
                timerInterval !== null
            ) {

                return;

            }


            if (timeLeft <= 0) {

                timeLeft =
                    TIMER_START;

                updateTimer();

            }


            document.getElementById(
                "timerMessage"
            ).innerText = "";


            timerInterval =
                setInterval(function() {

                    timeLeft--;

                    updateTimer();

                    saveTimer();


                    if (timeLeft <= 0) {

                        clearInterval(
                            timerInterval
                        );

                        timerInterval = null;

                        completeStudySession();

                        document.getElementById(
                            "timerMessage"
                        ).innerText =
                            "🎉 Study session completed!";

                        timeLeft =
                            TIMER_START;

                        saveTimer();

                    }

                }, 1000);

        }


        function pauseTimer() {

            if (
                timerInterval !== null
            ) {

                clearInterval(
                    timerInterval
                );

                timerInterval = null;

            }


            saveTimer();


            document.getElementById(
                "timerMessage"
            ).innerText =
                "⏸️ Study timer paused.";

        }


        function resetTimer() {

            clearInterval(
                timerInterval
            );

            timerInterval = null;

            timeLeft =
                TIMER_START;

            saveTimer();

            updateTimer();


            document.getElementById(
                "timerMessage"
            ).innerText = "";

        }


        function completeStudySession() {

            let sessions =
                getData(
                    "studySessions",

            let minutes =
                getData(

                    0

            sessions++;

            minutes += 5;


            saveData(
                "studySessions",
                sessions
            );

            saveData(
                "studyMinutes",
                minutes
            );


            updateDashboard();

        }

        function addNote() {

            let title =
                document.getElementById(
                    "noteTitle"
                ).value.trim();


            let text =
                document.getElementById(
                    "noteText"
                ).value.trim();


            if (
                title === "" ||
                text === ""
            ) {

                alert(
                    "Please enter both note title and note text."
                );

                return;

            }


            let notes =
                getData(
                    "myStudyNotes",
                    []
                );


            notes.push({

                title: title,

                text: text

            });


            saveData(
                "myStudyNotes",
                notes
            );


            document.getElementById(
                "noteTitle"
            ).value = "";


            document.getElementById(
                "noteText"
            ).value = "";


            displayNotes();

        }


        function displayNotes(
            searchText = ""
        ) {

            let notes =
                getData(
                    "myStudyNotes",
                    []
                );


            let notesList =
                document.getElementById(
                    "notesList"
                );


            notesList.innerHTML = "";


            let found =
                0;


            notes.forEach(
                function(note, index) {

                    let combined =
                        (
                            note.title +
                            " " +
                            note.text
                        ).toLowerCase();


                    if (
                        !combined.includes(
                            searchText.toLowerCase()
                        )
                    ) {

                        return;

                    }


                    found++;


                    let card =
                        document.createElement(
                            "div"
                        );


                    card.className =
                        "note-card";


                    let title =
                        document.createElement(
                            "h3"
                        );


                    title.innerText =
                        note.title;


                    let text =
                        document.createElement(
                            "p"
                        );


                    text.innerText =
                        note.text;


                    let deleteButton =
                        document.createElement(
                            "button"
                        );


                    deleteButton.innerText =
                        "🗑️ Delete";


                    deleteButton.onclick =
                        function() {

                            deleteNote(index);

                        };


                    card.appendChild(title);

                    card.appendChild(text);

                    card.appendChild(
                        deleteButton
                    );


                    notesList.appendChild(
                        card
                    );

                }
            );


            let message =
                document.getElementById(
                    "noNotesMessage"
                );


            if (found === 0) {

                message.style.display =
                    "block";

            } else {

                message.style.display =
                    "none";

            }

        }

        function loadTopics() {

            let completed =
                getData(
                    "completedTopics",
                    []
                );


            let checkboxes =
                document.querySelectorAll(
                    ".topic-checkbox"
                );


            checkboxes.forEach(
                function(box) {

                    box.checked =
                        completed.includes(
                            box.value
                        );

                    box.addEventListener(
                        "change",
                        saveTopics
                    );

                }
            );


            updateTopicProgress();

        }


        function saveTopics() {

            let completed = [];


            document.querySelectorAll(
                ".topic-checkbox"
            ).forEach(
                function(box) {

                    if (box.checked) {

                        completed.push(
                            box.value
                        );

                    }

                }
            );


            saveData(
                "completedTopics",
                completed
            );


            updateTopicProgress();

        }


        function updateTopicProgress() {

            let boxes =
                document.querySelectorAll(
                    ".topic-checkbox"
                );


            let completed =
                document.querySelectorAll(
                    ".topic-checkbox:checked"
                ).length;


            let total =
                boxes.length;


            let percentage =
                total === 0
                ? 0
                : (completed / total) * 100;


            document.getElementById(
                "topicsCompleted"
            ).innerText =
                completed;


            document.getElementById(
                "topicProgressBar"
            ).style.width =
                percentage + "%";

        }

            if (
                currentQuestion <
                mcqQuestions.length - 1
            ) {

                currentQuestion++;

                showQuestion(
                    currentQuestion
                );

            }

        }


        function previousQuestion() {

            if (
                currentQuestion > 0
            ) {

                currentQuestion--;

                showQuestion(
                    currentQuestion
                );

            }

        }

        function checkAnswers() {

            let correctAnswers = {

                q1: "B",
                q2: "B",
                q3: "A",
                q4: "C",
                q5: "C",
                q6: "A",
                q7: "C",
                q8: "B",
                q9: "B",
                q10: "C"

            };


            let score = 0;


            for (
                let question in correctAnswers
            ) {

                let selected =
                    document.querySelector(
                        'input[name="' +
                        question +
                        '"]:checked'
                    );


                if (
                    selected &&
                    selected.value ===
                    correctAnswers[question]
                ) {

                    score++;

                }

            }


            let wrong =
                10 - score;


            let percentage =
                score * 10;


            let performance = "";


            if (score === 10) {

                performance =
                    "🏆 Excellent! Perfect Score!";

            }

            else if (score >= 8) {

                performance =
                    "🌟 Excellent work! Keep it up!";

            }

            else if (score >= 6) {

                performance =
                    "👍 Good Job! Keep improving!";

            }

            else if (score >= 4) {

                performance =
                    "📚 Good effort! Keep practicing!";

            }

            else {

                performance =
                    "💪 Don't Give Up! Try Again!";

            }


            document.getElementById(
                "finalScore"
            ).innerText =
                "🎯 Score: " +
                score +
                " / 10";


            document.getElementById(
                "finalDetails"
            ).innerText =
                "✅ Correct: " +
                score +
                "   ❌ Wrong: " +
                wrong +
                "   📊 Percentage: " +
                percentage +
                "%";


            document.getElementById(
                "finalPerformance"
            ).innerText =
                performance;


            document.getElementById(
                "resultBox"
            ).style.display =
                "block";


            /* SAVE MCQ DATA */

            let previousBest =
                getData(
                    "bestScore",
                    0
                );


            let totalSolved =
                getData(
                    "mcqsSolved",
                    0
                );


            totalSolved += 10;


            if (
                score >
                previousBest
            ) {

                previousBest =
                    score;

            }


            saveData(
                "bestScore",
                previousBest
            );


            saveData(
                "lastScore",
                score
            );


            saveData(
                "lastPercentage",
                percentage
            );


            saveData(
                "mcqsSolved",
                totalSolved
            );


            updateDashboard();

        }

        function updateDailyGoal(
            totalMinutes
        ) {

            let goalMinutes =
                120;


            let completed =
                Math.min(
                    totalMinutes,
                    goalMinutes
                );


            let remaining =
                goalMinutes -
                completed;


            let completedHours =
                completed / 60;


            let remainingHours =
                remaining / 60;


            let percentage =
                (completed /
                goalMinutes) * 100;


            document.getElementById(
                "goalCompleted"
            ).innerText =
                "Completed: " +
                completedHours.toFixed(2) +
                " Hours";


            document.getElementById(
                "goalRemaining"
            ).innerText =
                "Remaining: " +
                remainingHours.toFixed(2) +
                " Hours";


            document.getElementById(
                "dailyProgressBar"
            ).style.width =
                percentage + "%";

        }
